dvicode-server - SystemVerilog language server (DVICode)

USAGE:
  dvicode-server [--stdio]                      LSP over stdin/stdout (default)
  dvicode-server --listen HOST:PORT             LSP over TCP for one client. With a token in
                                                $DVICODE_TOKEN (or --token T, which other users
                                                can see in the process list), the client must
                                                send "DVICODE-TOKEN <T>\n" first. Without one,
                                                anyone who can connect can read and write this
                                                account's files: use a token on shared machines.
  dvicode-server check <dir> [--build FILE] [-q] [--max N]
                                                Compile a project and print its problems
                                                (--build auto: auto-configure).
                                                Queries: --goto/--hover/--refs/--impl/--sig/
                                                --complete/--calls F:LINE:COL, --outline F,
                                                --fixes F, --hier TOP|-, --uvm TEST|-,
                                                --symbols Q, --expand F:LINE:COL, --files -
                                                --queries FILE: one query per line; panics
                                                are reported and each query is timed
                                                --query-timeout SECS: exit 124 when one
                                                query runs longer
                                                --fuzz-edits N [--seed S]: make N random
                                                edits and compare each incremental update
                                                with a full build; print reproducers of
                                                differences (exit 1). --fuzz-samples K,
                                                --fuzz-findings M, --fuzz-verbose;
                                                --fuzz-disk also writes files (use a copy)
  dvicode-server --version

OPTIONS:
  --log FILE          Write the log to FILE instead of stderr
  --log-level LEVEL   error | warn | info | debug | trace (default info)
